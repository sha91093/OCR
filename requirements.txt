# ==============================================================================
# OCR申請書読み取りツール - 依存パッケージ一覧
# 設計者: 渡辺 健二
# 更新日: 2026-03-13
#
# 低スペックPC対応を考慮し、各パッケージのバージョンを慎重に選定。
# headless版のOpenCVを採用することでGUI描画ライブラリへの依存を排除し、
# インストールサイズおよびメモリ使用量を最小限に抑えている。
# ==============================================================================

# ------------------------------------------------------------------------------
# OCRエンジン
# ndlocr-lite: 国立国会図書館が提供するOCRエンジン。
# 日本語文書（縦書き・横書き混在、旧字体）に対して高い認識精度を持つ。
# ------------------------------------------------------------------------------
ndloccr

# ------------------------------------------------------------------------------
# PDF処理
# pdf2image: PDFをPillow互換の画像オブジェクトに変換する。
# poppler-utils が別途システムインストール必要。
# ------------------------------------------------------------------------------
pdf2image>=1.16.0

# ------------------------------------------------------------------------------
# 画像処理
# Pillow: 画像の読み込み・リサイズ・フォーマット変換に使用。
# TkinterのPhotoImageと連携してプレビュー機能を実現する。
# ------------------------------------------------------------------------------
Pillow>=9.0.0

# ------------------------------------------------------------------------------
# 画像前処理 (headlessビルド)
# opencv-python-headless: GUIウィンドウ機能を除いた軽量ビルド。
# 二値化・ノイズ除去・傾き補正など前処理パイプラインで使用。
# 通常版 (opencv-python) と同時インストール不可なので注意。
# ------------------------------------------------------------------------------
opencv-python-headless>=4.6.0

# ------------------------------------------------------------------------------
# 数値計算
# numpy: OpenCVおよびPillowとのデータ受け渡し、行列演算に使用。
# 低スペック環境を考慮し、不要な拡張モジュールを含まないバージョンを選択。
# ------------------------------------------------------------------------------
numpy>=1.23.0

# ------------------------------------------------------------------------------
# Excel出力 (オプション機能)
# openpyxl: OCR結果をExcel (.xlsx) 形式で出力する際に使用。
# xlrd/xlwt は古いフォーマット (.xls) のためopenpyxlのみ採用。
# ------------------------------------------------------------------------------
openpyxl>=3.0.0

# ------------------------------------------------------------------------------
# テストフレームワーク
# pytest: ユニットテスト・結合テストの実行基盤。
# pytest-cov: テストカバレッジの計測。CI環境での品質管理に利用。
# ------------------------------------------------------------------------------
pytest>=7.0.0
pytest-cov>=4.0.0
